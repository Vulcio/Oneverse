import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

<CardGrid>
  <Card title="Chants" icon="document">
    Chaque chant avec ses accords, ses crédits, ses fichiers et ses enregistrements. Des grilles modifiées avec les accords juste au-dessus des paroles.
  </Card>
  <Card title="Versions" icon="setting">
    La façon dont votre groupe joue un chant - son ordre, sa tonalité, son capo, des accords changés - sans toucher au chant lui-même.
  </Card>
  <Card title="Listes de chants" icon="list-format">
    Les chants d'un culte ou d'un concert, chacun dans sa tonalité et sa version. Partagez-les par lien avec des musiciens invités.
  </Card>
  <Card title="Votre propre affichage" icon="seti:eye">
    Masquez les accords trop rapides pour vous, simplifiez-les, lisez-les en solfège ou en formes de capo. La grille des autres ne change pas.
  </Card>
</CardGrid>
